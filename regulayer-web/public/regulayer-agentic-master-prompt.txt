<python_trace_context_manager>
```python
from regulayer import trace
from regulayer.errors import GovernanceDeclinedError

def user_inference_function(user_query: str, user_email: str):
    # 1. INVOKE THE SDK CONTEXT MANAGER
    try:
        with trace(
            system="customer_support_bot",
            decision_type="generation",
            risk_level="high",  
            hidden_fields=["user_email"], # Hashes PII before recording
            zkp_salt=os.environ.get("ZKP_SECRET_SALT", "default_salt")
        ) as t:
            
            # 2. LOG THE INPUT
            input_context = {"query": user_query, "user_email": user_email}
            t.set_input(input_context)
            
            # 3. RUN THE ACTUAL LLM
            response = my_llm_call(input_context)
            output_context = {"response": response}
            
            # 4. LOG THE OUTPUT
            t.set_output(output_context)
            
            # CONTEXT EXIT: Records to WORM. Will pause thread here if Governance triggers.
            
        return output_context
    except GovernanceDeclinedError as e:
        # 5. HANDLE POLICY REJECTION
        return {"error": "This AI action was explicitly blocked by Compliance."}
```
</python_trace_context_manager>

<node_promises>
```typescript
import { createClient, GovernanceDeclinedError } from '@regulayer/sdk';

const regulayer = createClient({ apiKey: process.env.REGULAYER_API_KEY });

export async function askAI(userData: any) {
    const aiOutput = await runOpenAIMethod(userData);
    
    try {
        const result = await regulayer.recordDecision({
            system: 'candidate-screening',
            riskLevel: 'high',
            decisionType: 'classification',
            input: userData,
            output: aiOutput,
            hiddenFields: ['email', 'ssn'], // Protect PII!
            zkpSalt: process.env.ZKP_SECRET_SALT
        });
        return aiOutput;
    } catch (err) {
        if (err instanceof GovernanceDeclinedError) {
             return "Compliance execution halt.";
        }
        throw err;
    }
}
```
